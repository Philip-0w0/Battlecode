cmake_minimum_required(VERSION 3.20)
project(battlecode_native_tools LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(cam-tools cam_tools.cpp)
add_executable(core-tests tests/core_tests.cpp)
add_executable(analysis-name-tests analysis_name_tests.cpp)
if(MINGW)
  target_link_options(cam-tools PRIVATE -static)
  target_link_options(core-tests PRIVATE -static)
  target_link_options(analysis-name-tests PRIVATE -static)
endif()
enable_testing()
add_test(NAME strategy_parity COMMAND core-tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/python-parity.json")
add_test(NAME summary_names COMMAND analysis-name-tests)
